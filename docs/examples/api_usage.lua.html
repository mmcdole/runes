api_usage.lua

--[[
Runes MUD Client - API Usage Example

This example demonstrates different ways to use the Runes API.
]]

-- Method 1: Use the full namespace (most verbose but clearest)
runes.output("Method 1: Using full namespace")

runes.trigger.add("hp_watch", "HP: (%d+)/(%d+)", function(matches, line)
    local current = tonumber(matches[1])
    local max = tonumber(matches[2])
    if current < max * 0.3 then
        runes.output("LOW HEALTH WARNING!")
    end
end)

runes.alias.add("^hp$", "look at health")

-- Method 2: Using local variables (recommended)
runes.output("\nMethod 2: Using local variables")

local trigger = runes.trigger
local alias = runes.alias

trigger.add("mana_watch", "MP: (%d+)/(%d+)", function(matches, line)
    local current = tonumber(matches[1])
    local max = tonumber(matches[2])
    if current < max * 0.3 then
        runes.output("LOW MANA WARNING!")
    end
end)

alias.add("^mp$", "look at mana")

-- Method 3: Using the import helper (creates globals)
runes.output("\nMethod 3: Using the import helper")

-- Import modules into global scope
runes.import("events", "trigger", "alias")

-- Now events, trigger, and alias are available globally
events.add("custom_event", function(data)
    runes.output("Custom event received: " .. data.message)
end)

-- Trigger the custom event
events.emit("custom_event", { message = "Hello from the event system!" })

trigger.add("gold_watch", "Gold: (%d+)", function(matches, line)
    local gold = tonumber(matches[1])
    if gold > 1000 then
        runes.output("You're rich!")
    end
end)

-- Clean up our example triggers
runes.output("\nCleaning up example triggers")
runes.trigger.clear()
